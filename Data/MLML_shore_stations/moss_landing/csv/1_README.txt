The flags are appended to a list for each variable. The maximum flag value is applied to the datapoint.\r\n\r\n\r\ndef sensorRange( val , t_sensor_min , t_sensor_max ):\r\n    # val = current data point\r\n    # t_sensor_min = minimum scaled sensor value\r\n    # t_sensor_max = maximum scaled sensor value\r\n    if val < t_sensor_min or val > t_sensor_max:\r\n        flag = 4\r\n    elif math.isnan(val) == True:\r\n        flag = 9\r\n    else:\r\n        flag = 1\r\n    return flag\r\n\r\ndef userRange( val , t_user_min , t_user_max ):\r\n    # val = current data point\r\n    # t_user_min = minimum scaled sensor value\r\n    # t_user_max = maximum scaled sensor value\r\n    if val < t_user_min or val > t_user_max:\r\n        flag = 3\r\n    elif math.isnan(val) == True:\r\n        flag = 9\r\n    else:\r\n        flag = 1\r\n    return flag\r\n\r\ndef flatLine( val , comp , eps , rep_cnt_fail=6 , rep_cnt_suspect=3 ):\r\n    # val = current datapoint\r\n    # comp = comparison values (n). Must be n>rep_cnt_fail\r\n    # eps = tolerance value\r\n    # rep_cnt_fail = number of repeated equal values to fail (default=6)\r\n    # rep_cnt_suspect = number of repeated equal values to suspect (default=3)\r\n    fail=rep_cnt_fail\r\n    suspect=rep_cnt_suspect\r\n    eps=eps\r\n    if len(comp) < fail:\r\n        raise ValueError("Not enough values to compare. Number of values in comp must be greater than rep_cnt_fail and rep_cnt_suspect.")\r\n    \r\n\r\n    count = 0\r\n    flag=[]\r\n    for i in np.arange(len(comp)):\r\n        test = abs(val - comp[i])\r\n        if test<=eps:\r\n            count+= 1\r\n    if count >= suspect:\r\n        flag.append(3)\r\n\r\n    count = 0\r\n    for i in np.arange(len(comp)):\r\n        test = abs(val - comp[i])\r\n        if test<=eps:\r\n            count+= 1\r\n    if count >= fail:\r\n        flag.append(4)\r\n\r\n    else:\r\n        flag.append(1)\r\n\r\ndef spike_ref(current_temp, current_time, compList, compTimes, thrshld_low, thrshld_high, time_gap_threshold):\r\n    # Ensure compList has sufficient length\r\n    if len(compList) < 2 or len(compTimes) < 2:\r\n        return 2  # Not enough data\r\n\r\n    # Calculate the time gap\r\n    last_time = compTimes[0]\r\n    time_gap = current_time - last_time\r\n\r\n    # Check if time gap exceeds the threshold\r\n    if time_gap > time_gap_threshold:\r\n        return 2  # Low flag due to time gap\r\n\r\n    # Calculate the baseline mean from historical data\r\n    mean = np.nanmean(compList[:3])\r\n\r\n    # Calculate the spike reference value (difference between current temp and baseline mean)\r\n    spk_ref = abs(current_temp - mean)\r\n\r\n    # Determine the flag based on thresholds\r\n    if spk_ref >= thrshld_high:\r\n        return 4  # Spike detected\r\n    elif spk_ref >= thrshld_low:\r\n        return 3  # Near spike\r\n    else:\r\n        return 1  # No spike\r\n\r\n\r\n\r\n'
